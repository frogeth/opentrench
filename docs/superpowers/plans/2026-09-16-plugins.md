# Plugins Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users drop a one-file JavaScript plugin into a git-ignored `plugins` folder, approve it after a warning, and have it post messages into the feed, fetch sites through the app's own logins, and optionally draw a column, all through a sandboxed, versioned `ot` API.

**Architecture:** The backend owns the plugin folder, manifests, approval hashes, per-plugin storage, a rate-capped feed input, and a fetch proxy that hands site-authenticated requests to the desktop shell over a local callback. The frontend runs each enabled plugin in a sandboxed iframe (opaque origin, `allow-scripts` only) whose only door is a postMessage bridge; a pure `routeCall` function applies permissions and maps calls to app actions. The shell keeps one cookie partition per site and opens sign-in windows.

**Tech Stack:** Node 22 / TypeScript / express (backend, vitest), React 18 / Vite (frontend; vitest + jsdom added for the router test), Electron 33 (shell).

Spec: `docs/superpowers/specs/2026-09-16-plugins-design.md`. Never mention any real site in code, docs, examples or commits; the example site is `example.com`.

```ts
// backend/src/plugins/manifest.ts
/** What a plugin declares at the top of its file. The block must be plain JSON so the app can read it without running anything. */
export interface PluginManifest {
  id: string;
  name: string;
  version: string;
  /** opentrench plugin API major version the plugin was written for */
  api: number;
  /** https origins the plugin may fetch through the user's login */
  sites: string[];
  permissions: PluginPermission[];
  /** true when the plugin draws a column */
  ui: boolean;
  description: string;
}
export type PluginPermission = 'feed:write' | 'storage' | 'actions';
export const PERMISSIONS: PluginPermission[] = ['feed:write', 'storage', 'actions'];
/** The API major this build serves. Append-only within a major; bump when anything is renamed or removed. */
export const PLUGIN_API_VERSION = 1;
const ID_RE = /^[a-z0-9][a-z0-9-]{1,39}$/;
const MAX_FILE = 512 * 1024;

/** Finds `export const manifest =`, takes the `{…}` that follows (string-aware brace matching), parses it as JSON, validates. */
export function extractManifest(source: string): PluginManifest {
  if (source.length > MAX_FILE) throw new Error('plugin file is too large (512 KB max)');
  const at = source.search(/export\s+const\s+manifest\s*=\s*\{/);
  if (at < 0) throw new Error('no manifest: the file must start with `export const manifest = { … };`');
  const start = source.indexOf('{', at);
  let depth = 0;
  let inStr: string | null = null;
  let end = -1;
  for (let i = start; i < source.length; i++) {
    const ch = source[i];
    if (inStr) {
      if (ch === '\\') i++;
      else if (ch === inStr) inStr = null;
      continue;
    }
    if (ch === '"' || ch === "'") inStr = ch;
    else if (ch === '{') depth++;
    else if (ch === '}') {
      depth--;
      if (depth === 0) {
        end = i;
        break;
      }
    }
  }
  if (end < 0) throw new Error('manifest block never closes');
  let raw: any;
  try {
    raw = JSON.parse(source.slice(start, end + 1));
  } catch {
    throw new Error('manifest is not valid JSON (quoted keys, no comments, no trailing commas)');
  }
  return validateManifest(raw);
}

export function validateManifest(raw: any): PluginManifest {
  if (!raw || typeof raw !== 'object') throw new Error('manifest must be an object');
  const id = String(raw.id ?? '');
  if (!ID_RE.test(id)) throw new Error('manifest id must be 2–40 chars of a-z, 0-9 and dashes');
  const name = String(raw.name ?? '').trim().slice(0, 60);
  if (!name) throw new Error('manifest needs a name');
  const version = String(raw.version ?? '').trim().slice(0, 20);
  if (!/^\d+\.\d+\.\d+/.test(version)) throw new Error('manifest version must look like 1.2.3');
  const api = Number(raw.api);
  if (!Number.isInteger(api) || api < 1) throw new Error('manifest api must be a positive integer');
  if (api > PLUGIN_API_VERSION) throw new Error(`this plugin needs plugin api ${api}; this build serves ${PLUGIN_API_VERSION} — update opentrench`);
  const sites = Array.isArray(raw.sites) ? raw.sites.map(String) : [];
  for (const s of sites) {
    let u: URL;
    try {
      u = new URL(s);
    } catch {
      throw new Error(`sites: ${s} is not a URL`);
    }
    if (u.protocol !== 'https:' || u.pathname !== '/' || u.search || u.hash) throw new Error(`sites: ${s} must be an https origin like https://example.com`);
  }
  const permissions = Array.isArray(raw.permissions) ? raw.permissions.map(String) : [];
  for (const p of permissions) if (!PERMISSIONS.includes(p as PluginPermission)) throw new Error(`unknown permission "${p}" (allowed: ${PERMISSIONS.join(', ')})`);
  return { id, name, version, api, sites: [...new Set(sites.map((s: string) => new URL(s).origin))], permissions: [...new Set(permissions)] as PluginPermission[], ui: raw.ui === true, description: String(raw.description ?? '').slice(0, 300) };
}
```

```ts
// backend/src/plugins/state.ts
import fs from 'node:fs';
import path from 'node:path';

export interface PluginRecord {
  storage: Record<string, unknown>;
  settings: Record<string, unknown>;
  chats: Record<string, string>;
}
const EMPTY = (): PluginRecord => ({ storage: {}, settings: {}, chats: {} });
const MAX_STORAGE_BYTES = 256 * 1024;
/** Coalesce a burst of writes… */
const DEBOUNCE_MS = 200;
/** …but never let a steady stream of them hold the file back longer than this. */
const MAX_DEBOUNCE_MS = 2000;

/** `plugins-state.json` next to the config: per-plugin storage, settings values and the chats it has posted. Debounced writes. */
export class PluginState {
  private data: Record<string, PluginRecord> = {};
  private timer: NodeJS.Timeout | undefined;
  /** when the oldest unsaved change came in, 0 when everything is on disk */
  private firstDirtyAt = 0;
  constructor(private file: string) {
    try {
      const raw = JSON.parse(fs.readFileSync(file, 'utf8'));
      for (const [id, r] of Object.entries<any>(raw ?? {})) this.data[id] = { storage: r?.storage ?? {}, settings: r?.settings ?? {}, chats: r?.chats ?? {} };
    } catch {
      /* first run */
    }
  }
  /** Read-only view. An id with nothing stored reads as empty without being written down. */
  read(id: string): PluginRecord {
    return this.data[id] ?? EMPTY();
  }
  private ensure(id: string): PluginRecord {
    return (this.data[id] ??= EMPTY());
  }
  setStorage(id: string, key: string, value: unknown): void {
    const r = this.ensure(id);
    const prev = r.storage[key];
    const had = key in r.storage;
    if (value === undefined) delete r.storage[key];
    else r.storage[key] = value;
    if (Buffer.byteLength(JSON.stringify(r.storage)) > MAX_STORAGE_BYTES) {
      if (had) r.storage[key] = prev;
      else delete r.storage[key];
      throw new Error(`plugin storage is full (${MAX_STORAGE_BYTES / 1024} KB)`);
    }
    this.save();
  }
  setSettings(id: string, values: Record<string, unknown>): void {
    this.ensure(id).settings = values;
    this.save();
  }
  noteChat(id: string, chatId: string, name: string): void {
    this.ensure(id).chats[chatId] = name;
    this.save();
  }
  forget(id: string): void {
    delete this.data[id];
    this.save();
  }
  /** Write now (tests, shutdown). */
  flush(): void {
    clearTimeout(this.timer);
    this.timer = undefined;
    this.firstDirtyAt = 0;
    try {
      fs.mkdirSync(path.dirname(this.file), { recursive: true });
      const tmp = `${this.file}.tmp`;
      fs.writeFileSync(tmp, JSON.stringify(this.data, null, 2), { mode: 0o600 });
      fs.renameSync(tmp, this.file);
      try {
        // The mode above only applies when the tmp file is created; rename carries it over, but a
        // file left world-readable by an older version keeps its mode, so tighten it explicitly.
        // Windows has no real POSIX mode bits and can throw here — nothing to tighten there.
        fs.chmodSync(this.file, 0o600);
      } catch {
        /* best-effort */
      }
    } catch (e: any) {
      console.warn('[plugins] state save failed', e?.message ?? e);
    }
  }
  private save(): void {
    const now = Date.now();
    if (!this.firstDirtyAt) this.firstDirtyAt = now;
    // A plugin writing every tick would reset the debounce forever and lose everything on a crash.
    if (now - this.firstDirtyAt >= MAX_DEBOUNCE_MS) {
      this.flush();
      return;
    }
    clearTimeout(this.timer);
    this.timer = setTimeout(() => this.flush(), DEBOUNCE_MS);
    this.timer.unref?.();
  }
}
```

```ts
// backend/src/plugins/registry.ts
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';
import type { PluginInfo } from '../types.js';
import { extractManifest, ManifestError, MAX_FILE, MAX_FILE_MESSAGE, type PluginManifest } from './manifest.js';
import type { PluginState } from './state.js';

interface Loaded {
  id: string;
  file: string;
  hash: string;
  manifest?: PluginManifest;
  error?: string;
}
export interface LogLine {
  ts: number;
  level: 'info' | 'warn' | 'error';
  text: string;
}
/** The slice of ConfigStore the registry needs (keeps tests free of the real store). */
export interface PluginConfig {
  get(): { plugins: Record<string, { enabled: boolean; approvedHash?: string }> };
  update(fn: (c: { plugins: Record<string, { enabled: boolean; approvedHash?: string }> }) => void): void;
}
const LOG_MAX = 200;

/** An id for a broken file that no other entry holds: its name, else the name plus a slice of its hash, else a counter. */
function freeId(taken: Map<string, Loaded>, base: string, hash: string): string {
  if (!taken.has(base)) return base;
  if (hash) {
    const withHash = `${base}-${hash.slice(0, 6)}`;
    if (!taken.has(withHash)) return withHash;
  }
  for (let n = 2; ; n++) {
    const numbered = `${base}-${n}`;
    if (!taken.has(numbered)) return numbered;
  }
}

/** The plugins folder: what is in it, whether each file is approved and enabled, and what each plugin has been doing. */
export class PluginRegistry {
  private loaded = new Map<string, Loaded>();
  private logLines = new Map<string, LogLine[]>();
  /** sites with a shell session; filled in by the shell link */
  signedIn: () => string[] = () => [];
  onChange: () => void = () => {};

  constructor(readonly dir: string, private state: PluginState, private cfg: PluginConfig) {}

  /**
   * Rescan the folder. Errors are per file: a broken plugin never hides the others.
   * Two passes, because a broken file's id is guessed from its name and could collide with a real
   * plugin's: valid plugins claim their manifest id first, then the broken ones take what is left.
   */
  load(): void {
    fs.mkdirSync(this.dir, { recursive: true });
    const next = new Map<string, Loaded>();
    const broken: Loaded[] = [];
    for (const name of fs.readdirSync(this.dir).filter((f) => f.endsWith('.js')).sort()) {
      const file = path.join(this.dir, name);
      const stem = name.replace(/\.js$/, '');
      const fallbackId = stem.toLowerCase().replace(/[^a-z0-9-]/g, '-').slice(0, 40) || 'plugin';
      let bytes: Buffer;
      try {
        // lstat, not stat: a symlink here would otherwise let the folder reach any file on the box.
        const st = fs.lstatSync(file);
        if (!st.isFile()) throw new Error('not a regular file');
        if (st.size > MAX_FILE) throw new ManifestError(MAX_FILE_MESSAGE);
        bytes = fs.readFileSync(file);
      } catch (e: any) {
        broken.push({ id: fallbackId, file, hash: '', error: e?.message ?? String(e) });
        continue;
      }
      // Hash the bytes, not the decoded text: two different files can decode to the same string.
      const hash = crypto.createHash('sha256').update(bytes).digest('hex');
      const source = bytes.toString('utf8');
      try {
        const manifest = extractManifest(source);
        if (manifest.id !== stem) throw new Error(`file name must be ${manifest.id}.js`);
        next.set(manifest.id, { id: manifest.id, file, hash, manifest });
      } catch (e: any) {
        broken.push({ id: fallbackId, file, hash, error: e?.message ?? String(e) });
      }
    }
    for (const bad of broken) {
      const id = freeId(next, bad.id, bad.hash);
      next.set(id, { ...bad, id });
    }
    this.loaded = next;
    this.onChange();
  }

  list(): PluginInfo[] {
    const cfg = this.cfg.get().plugins;
    const signed = new Set(this.signedIn());
    return [...this.loaded.values()].map((p) => {
      const c = cfg[p.id];
      const needsApproval = !c?.approvedHash || c.approvedHash !== p.hash;
      return {
        id: p.id,
        file: path.basename(p.file),
        hash: p.hash,
        manifest: p.manifest && { ...p.manifest, sites: [...p.manifest.sites], permissions: [...p.manifest.permissions] },
        error: p.error,
        enabled: !!c?.enabled && !needsApproval && !p.error,
        needsApproval,
        // Only a real plugin gets a state record; a junk file must not add one to plugins-state.json.
        chats: p.manifest ? { ...this.state.read(p.id).chats } : {},
        signedIn: (p.manifest?.sites ?? []).filter((s) => signed.has(s)),
      };
    });
  }
  get(id: string): PluginInfo | undefined {
    return this.list().find((p) => p.id === id);
  }
  manifest(id: string): PluginManifest {
    const p = this.loaded.get(id);
    if (!p?.manifest) throw new Error(p?.error ?? 'unknown plugin');
    return p.manifest;
  }
  /**
   * The file's text, for the iframe to import. Only an enabled plugin's code leaves the folder, and
   * only the exact bytes the user approved: the file is re-hashed here, because the scan that
   * approved it may be seconds or days old and the file can have been swapped since.
   */
  code(id: string): string {
    const p = this.get(id);
    if (!p?.enabled) throw new Error('plugin is not enabled');
    const entry = this.loaded.get(id)!;
    const bytes = fs.readFileSync(entry.file);
    if (crypto.createHash('sha256').update(bytes).digest('hex') !== entry.hash) {
      this.load(); // so the next list() shows it as needing approval again
      throw new Error('the plugin file changed on disk; approve this version first');
    }
    return bytes.toString('utf8');
  }
  approve(id: string): void {
    const p = this.loaded.get(id);
    if (!p?.manifest) throw new Error(p?.error ?? 'unknown plugin');
    this.cfg.update((c) => {
      c.plugins[id] = { ...(c.plugins[id] ?? { enabled: false }), approvedHash: p.hash };
    });
    this.onChange();
  }
  enable(id: string): void {
    const p = this.loaded.get(id);
    if (!p?.manifest) throw new Error(p?.error ?? 'unknown plugin');
    if (this.cfg.get().plugins[id]?.approvedHash !== p.hash) throw new Error('approve this version of the plugin first');
    this.cfg.update((c) => {
      c.plugins[id] = { ...c.plugins[id], enabled: true };
    });
    this.onChange();
  }
  disable(id: string): void {
    this.cfg.update((c) => {
      if (c.plugins[id]) c.plugins[id].enabled = false;
    });
    this.onChange();
  }
  /** Validate, then write `<id>.js`. Replacing an existing file drops its approval (the hash changes). */
  add(source: string): PluginInfo {
    const m = extractManifest(source);
    fs.mkdirSync(this.dir, { recursive: true });
    const target = path.join(this.dir, `${m.id}.js`);
    try {
      // Never write *through* a symlink someone planted in the folder: drop whatever is in the way
      // unless it is an ordinary file we are meant to replace.
      if (!fs.lstatSync(target).isFile()) fs.rmSync(target, { force: true, recursive: true });
    } catch {
      /* nothing there yet */
    }
    fs.writeFileSync(target, source, { mode: 0o600 });
    this.load();
    return this.get(m.id)!;
  }
  remove(id: string): void {
    const p = this.loaded.get(id);
    if (!p) throw new Error('unknown plugin');
    fs.rmSync(p.file, { force: true });
    this.cfg.update((c) => {
      delete c.plugins[id];
    });
    this.state.forget(id);
    this.logLines.delete(id);
    this.load();
  }
  noteChat(id: string, chatId: string, name: string): void {
    if (this.state.read(id).chats[chatId] === name) return;
    this.state.noteChat(id, chatId, name);
    this.onChange();
  }
  log(id: string, level: LogLine['level'], text: string): void {
    if (!this.loaded.has(id)) return;
    const lines = this.logLines.get(id) ?? [];
    lines.push({ ts: Date.now(), level, text: text.slice(0, 2000) });
    if (lines.length > LOG_MAX) lines.splice(0, lines.length - LOG_MAX);
    this.logLines.set(id, lines);
  }
  logs(id: string): LogLine[] {
    return [...(this.logLines.get(id) ?? [])];
  }
}
```

```ts
// backend/src/plugins/feed.ts
import type { FeedMessage } from '../types.js';
import type { PluginManifest } from './manifest.js';

/** What a plugin hands to `ot.feed.post`. */
export interface PluginPost {
  /** the plugin's own stable id for this item (dedupe) */
  id: string;
  /** the chat this belongs to, as the plugin names it; one plugin can post several chats */
  chat: string;
  author: string;
  text: string;
  ts?: number;
  avatar?: string;
  link?: string;
  attachments?: { url: string; kind?: 'image' | 'video' }[];
}
const TEXT_MAX = 4000;
const slug = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40) || 'chat';
const httpUrl = (u: unknown, what: string): string | undefined => {
  if (u === undefined || u === null || u === '') return undefined;
  const s = String(u);
  if (!/^https?:\/\//i.test(s) || s.length > 2048) throw new Error(`${what} must be an http(s) URL`);
  return s;
};

export function chatIdFor(pluginId: string, chat: string): string {
  return `plugin:${pluginId}:${slug(chat)}`;
}

/** Validates a post and shapes it as a feed message. Contract detection, hidden and repeat flags are the hub's job. */
export function toFeedMessage(pluginId: string, manifest: PluginManifest, p: PluginPost): FeedMessage {
  const id = String(p?.id ?? '').trim().slice(0, 120);
  if (!id) throw new Error('post needs an id');
  const chat = String(p?.chat ?? '').trim().slice(0, 60);
  if (!chat) throw new Error('post needs a chat name');
  const author = String(p?.author ?? manifest.name).trim().slice(0, 60) || manifest.name;
  const text = String(p?.text ?? '').slice(0, TEXT_MAX);
  if (!text.trim() && !(p.attachments?.length)) throw new Error('post needs text or an attachment');
  const ts = Number.isFinite(p?.ts) && (p.ts as number) > 0 ? Math.round(p.ts as number) : Date.now();
  const chatId = chatIdFor(pluginId, chat);
  const media = (p.attachments ?? []).slice(0, 8).map((a) => ({ kind: a.kind === 'video' ? ('video' as const) : ('image' as const), url: httpUrl(a.url, 'attachment url')! }));
  return {
    id: `${chatId}:${id}`,
    source: 'plugin',
    chatId,
    chatName: chat,
    author,
    avatar: httpUrl(p.avatar, 'avatar'),
    isBot: false,
    hidden: false,
    text,
    ts,
    contracts: [],
    repeat: false,
    link: httpUrl(p.link, 'link'),
    hasAttachment: media.length > 0,
    media: media.map((m) => ({ ...m, mime: m.kind === 'video' ? 'video/mp4' : 'image/*' })) as any,
  } as FeedMessage;
}

/** Sliding-window cap per plugin so a runaway loop cannot flood the feed. */
export class PostLimiter {
  private hits = new Map<string, number[]>();
  constructor(private max: number, private windowMs: number) {}
  take(pluginId: string, now = Date.now()): boolean {
    const arr = (this.hits.get(pluginId) ?? []).filter((t) => now - t < this.windowMs);
    if (arr.length >= this.max) {
      this.hits.set(pluginId, arr);
      return false;
    }
    arr.push(now);
    this.hits.set(pluginId, arr);
    return true;
  }
}
```

The proxy is the one place a plugin touches the network, so this class is written as a gate rather than a
convenience: redirects are followed by hand so every hop is re-checked, request and response headers are
filtered in both directions, addresses are resolved before a socket opens, and the shell's reply is checked
rather than trusted.

- [ ] **Step 1: Write the failing test**

`backend/src/plugins/shell.test.ts`, eleven cases against `new ShellLink(fetchImpl)` with a `vi.fn()` fetch
(and a stand-in dispatcher, since nothing reaches the network):

1. *without a shell* — `available()` false, `signedIn()` `[]`, `signIn()` rejects with 'desktop app', a plain
   fetch returns `{ status, headers, body, truncated: false }` and went out with `redirect: 'manual'` and a
   `dispatcher`.
2. *with a shell* — `hello(45678, 'tok')`, then `/status` and `/fetch` go to `http://127.0.0.1:45678` with
   `authorization: Bearer tok`, and `/fetch` carries `{ url, init }`.
3. *plain redirects* — follows a relative 302, refuses a hop to `127.0.0.1` with 'local address', refuses a
   loop with 'redirects', refuses a local URL outright, and truncates a 5 MB body to `BODY_MAX` with
   `truncated: true` and a `content-length` that matches what the plugin actually got.
4. *a shell that stops answering* — status degrades to `[]`, and a site fetch **fails** rather than going out

5. *headers* — `set-cookie`, `set-cookie2` and unlisted headers are dropped on both paths; `content-type`,

6. *credentials* — a cross-origin redirect drops `authorization`; a same-origin one keeps it; `cookie` and

7. *method rewrite* — a 303 turns POST into GET and drops the body and its `content-type`.
8. *schemes* — a `data:` Location, a `file:` URL, and an https→http Location all throw ('http and https only',
   'insecure redirect').
9. *DNS rebinding* — `makeSafeLookup` with an injected resolver: an answer of `127.0.0.1` comes back as an
   error containing 'local address'; a public answer is passed through unchanged, which is what gets dialled.
10. *shell reply shape* — a status of `'two hundred'` is refused with 'bad status'.
11. *hello trust* — a second `hello` is ignored while the first shell answers, accepted at once when a probe
    of the first goes unanswered, accepted after three consecutive transport failures, and `goodbye()` clears
    the link. Three oversized replies do **not** cost the shell its registration; three refused connections do.
    Two hellos fired at once queue: the second re-checks the link after the first has settled.
13. *the header gate* — a headers object that is not the global `Headers` class (undici's own, or any object
    with `entries()`) still passes content-type and etag through and still loses set-cookie.
12. *one undici* — with the default `fetchImpl` and a real `safeDispatcher()`, a failed lookup comes back as a
    DNS error and never as 'invalid onRequestStart method' (what Node's global fetch does with a foreign
    dispatcher).

```ts
  // undici's Headers and the global one are different classes, and an identity check drops every header.

  //   '/fetch' (envelope budget BODY_MAX * 3 — JSON escaping, not the braces, is what makes it big), then
  //   shapeShellResponse (integer status 100–599, headers
  //   through the allow-list, body must be text) — the shell is a separate process on its own release cycle.
  // otherwise → one AbortSignal.timeout for the whole chain, then up to MAX_REDIRECTS hops, each one:
  //   http:/https: only · isLocalHost refused · https→http refused ('insecure redirect') ·
  //   credentials dropped when the origin changes · 303 (and 301/302 on POST) rewritten to a bodyless GET ·
  //   a cross-origin 307/308 keeps its method but loses its body and content-type ·
  //   request header names must be RFC 9110 tokens ·
  //   dispatcher pins the resolved address · body capped · content-length set to what we really hand over.
```

```ts
// backend/src/plugins/api.ts
import { Router, type Request, type Response } from 'express';
import type { MessageHub } from '../hub.js';
import type { ServerEvent } from '../types.js';
import { PostLimiter, toFeedMessage, type PluginPost } from './feed.js';
import type { PluginRegistry } from './registry.js';
import type { PluginState } from './state.js';
import type { ShellLink } from './shell.js';
import { isLocalHost } from './hosts.js';

interface WatchConfig {
  get(): { pluginWatch: string[] };
  update(fn: (c: { pluginWatch: string[] }) => void): void;
}

/** `/api/plugins/*` and `/api/shell/hello`. Every plugin-facing route checks the plugin is enabled. */
export function createPluginsApi(reg: PluginRegistry, state: PluginState, hub: MessageHub, shell: ShellLink, cfg: WatchConfig): Router {
  const r = Router();
  const limiter = new PostLimiter(60, 60_000);
  const fail = (res: Response, status: number, message: string) => res.status(status).json({ error: message });
  const enabled = (req: Request, res: Response): string | null => {
    const id = String(req.params.id);
    const p = reg.get(id);
    if (!p) {
      fail(res, 404, 'unknown plugin');
      return null;
    }
    if (!p.enabled) {
      fail(res, 403, 'plugin is not enabled');
      return null;
    }
    return id;
  };
  const broadcast = () => hub.emit('event', { type: 'plugins', plugins: reg.list() } satisfies ServerEvent);
  reg.onChange = broadcast;
  reg.signedIn = () => signedInCache;
  let signedInCache: string[] = [];
  const refreshSignedIn = async () => {
    const sites = [...new Set(reg.list().flatMap((p) => p.manifest?.sites ?? []))];
    signedInCache = await shell.signedIn(sites);
    broadcast();
  };

  r.get('/plugins', (_req, res) => res.json(reg.list()));
  r.get('/plugins/shell', (_req, res) => res.json({ available: shell.available() }));
  r.post('/shell/hello', async (req, res) => {
    // Loopback callers only: this is the socket a plugin's site-authenticated fetches go out on.
    if (!isLocalHost(String(req.socket.remoteAddress ?? ''))) return fail(res, 403, 'local callers only');
    const port = Number(req.body?.port);
    const token = String(req.body?.token ?? '');
    if (!Number.isInteger(port) || port <= 0 || !token) return fail(res, 400, 'port and token');
    try {
      // hello is async: it probes the shell already registered (if any) before handing the link over,
      // and concurrent hellos queue, so the route waits rather than firing and forgetting.
      await shell.hello(port, token);
    } catch (e: any) {
      return fail(res, 400, e?.message ?? 'hello');
    }
    void refreshSignedIn();
    res.json({ ok: true });
  });
  r.post('/plugins/reload', (_req, res) => {
    reg.load();
    res.json(reg.list());
  });
  r.post('/plugins/add', (req, res) => {
    try {
      const source = typeof req.body?.source === 'string' ? req.body.source : '';
      if (!source) return fail(res, 400, 'source');
      res.json(reg.add(source));
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  r.post('/plugins/add-url', async (req, res) => {
    try {
      const url = String(req.body?.url ?? '');
      if (!/^https:\/\//i.test(url)) return fail(res, 400, 'an https URL to a .js file');
      const got = await fetch(url, { signal: AbortSignal.timeout(15_000) });
      if (!got.ok) return fail(res, 400, `fetch failed: ${got.status}`);
      res.json(reg.add(await got.text()));
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  r.post('/plugins/:id/approve', (req, res) => {
    try {
      reg.approve(String(req.params.id));
      res.json({ ok: true });
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  r.post('/plugins/:id/enable', (req, res) => {
    try {
      reg.enable(String(req.params.id));
      void refreshSignedIn();
      res.json({ ok: true });
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  r.post('/plugins/:id/disable', (req, res) => {
    reg.disable(String(req.params.id));
    res.json({ ok: true });
  });
  r.delete('/plugins/:id', (req, res) => {
    reg.remove(String(req.params.id));
    cfg.update((c) => {
      c.pluginWatch = c.pluginWatch.filter((k) => !k.startsWith(`plugin:${req.params.id}:`));
    });
    res.json({ ok: true });
  });
  r.get('/plugins/:id/code', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    res.setHeader('content-type', 'text/javascript; charset=utf-8');
    res.setHeader('cache-control', 'no-store');
    res.send(reg.code(id));
  });
  r.get('/plugins/:id/logs', (req, res) => res.json(reg.logs(String(req.params.id))));
  r.post('/plugins/:id/log', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    const level = req.body?.level === 'error' ? 'error' : req.body?.level === 'warn' ? 'warn' : 'info';
    reg.log(id, level, String(req.body?.text ?? ''));
    res.json({ ok: true });
  });

  // ---- feed
  r.post('/plugins/:id/post', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    const manifest = reg.manifest(id);
    if (!manifest.permissions.includes('feed:write')) return fail(res, 403, 'plugin lacks the feed:write permission');
    if (!limiter.take(id)) return fail(res, 429, 'too many posts (60 a minute)');
    let msg;
    try {
      msg = toFeedMessage(id, manifest, req.body as PluginPost);
    } catch (e: any) {
      return fail(res, 400, e?.message ?? String(e));
    }
    // noteChat first, and inside the try: it enforces the 32-chats-per-plugin cap and throws
    // ('too many chats (32 max)'), which must refuse the post with a 400 before anything reaches the hub.
    try {
      reg.noteChat(id, msg.chatId, msg.chatName);
    } catch (e: any) {
      return fail(res, 400, e?.message ?? String(e));
    }
    cfg.update((c) => {
      if (!c.pluginWatch.includes(msg.chatId)) c.pluginWatch.push(msg.chatId);
    });
    hub.push(msg);
    res.json({ ok: true, id: msg.id });
  });
  r.post('/plugins/:id/patch', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    const msgId = String(req.body?.id ?? '');
    if (!msgId.startsWith(`plugin:${id}:`)) return fail(res, 400, 'not this plugin\'s message');
    const patch: Record<string, unknown> = {};
    if (typeof req.body?.text === 'string') patch.text = req.body.text.slice(0, 4000);
    hub.patch(msgId, patch);
    res.json({ ok: true });
  });
  r.put('/plugins/watch', (req, res) => {
    const key = String(req.body?.key ?? '');
    if (!/^plugin:[a-z0-9-]+:[a-z0-9-]+$/.test(key)) return fail(res, 400, 'key');
    cfg.update((c) => {
      c.pluginWatch = c.pluginWatch.filter((k) => k !== key);
      if (req.body?.on) c.pluginWatch.push(key);
    });
    res.json({ ok: true });
  });

  // ---- storage and settings
  r.get('/plugins/:id/storage', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    res.json(state.read(id).storage);
  });
  r.put('/plugins/:id/storage/:key', (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    if (!reg.manifest(id).permissions.includes('storage')) return fail(res, 403, 'plugin lacks the storage permission');
    try {
      state.setStorage(id, String(req.params.key).slice(0, 100), req.body?.value);
      res.json({ ok: true });
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  r.get('/plugins/:id/settings', (req, res) => res.json(state.read(String(req.params.id)).settings));
  r.put('/plugins/:id/settings', (req, res) => {
    const values = req.body?.values;
    if (!values || typeof values !== 'object') return fail(res, 400, 'values');
    state.setSettings(String(req.params.id), values);
    res.json({ ok: true });
  });

  // ---- network
  r.post('/plugins/:id/fetch', async (req, res) => {
    const id = enabled(req, res);
    if (!id) return;
    const url = String(req.body?.url ?? '');
    let u: URL;
    try {
      u = new URL(url);
    } catch {
      return fail(res, 400, 'url');
    }
    if (u.protocol !== 'https:' && u.protocol !== 'http:') return fail(res, 400, 'only http(s) URLs');
    if (isLocalHost(u.hostname)) return fail(res, 400, 'local addresses are off limits'); // the same rule the fetch path enforces
    const init = req.body?.init ?? {};
    const method = ['GET', 'POST', 'PUT', 'DELETE', 'PATCH', 'HEAD'].includes(String(init.method).toUpperCase()) ? String(init.method).toUpperCase() : 'GET';
    const headers: Record<string, string> = {};
    for (const [k, v] of Object.entries(init.headers ?? {})) if (/^[a-z0-9-]+$/i.test(k) && !/^(cookie|authorization-shell|host)$/i.test(k)) headers[k] = String(v).slice(0, 4000);
    const body = typeof init.body === 'string' ? init.body.slice(0, 1024 * 1024) : undefined;
    const viaShell = reg.manifest(id).sites.includes(u.origin);
    try {
      res.json(await shell.fetch(u.toString(), { method, headers, body }, viaShell));
    } catch (e: any) {
      fail(res, 502, e?.message ?? String(e));
    }
  });
  r.get('/plugins/:id/sites', async (req, res) => {
    const id = String(req.params.id);
    const sites = reg.manifest(id).sites;
    res.json({ sites, signedIn: await shell.signedIn(sites), available: shell.available() });
  });
  r.post('/plugins/:id/sites/signin', async (req, res) => {
    const id = String(req.params.id);
    const site = String(req.body?.site ?? '');
    if (!reg.manifest(id).sites.includes(site)) return fail(res, 400, 'not one of this plugin\'s sites');
    try {
      await shell.signIn(site);
      setTimeout(() => void refreshSignedIn(), 15_000).unref();
      res.json({ ok: true });
    } catch (e: any) {
      fail(res, 400, e?.message ?? String(e));
    }
  });
  return r;
}
```

- [x] **Step 1: `electron/shell-link.js`**

CommonJS like the rest of `electron/`. Exports `startShellLink({ log })` → `Promise<{ port, token, close() }>`,
`hello(backendUrl, link, log)`, `partitionFor(site)` (`persist:site-` + the first 16 hex of a sha256 of the *origin* — the origin because
two origins differing only in scheme or port are different logins, the hash because a host can hold characters
a partition name cannot and any substitution scheme that flattens them lets two hosts collide), and the four pure helpers the rules live in, so they can
be tested in plain Node: `isLocalHost`, `safeResponseHeaders`, `requestHeadersFor`, `nextHop`.

`require('electron')` is guarded: outside the app it either throws (not installed) or resolves to a *string*
(the npm package exports the path to the binary), so the module checks for `session`/`BrowserWindow` on the
result and keeps `electron = null` otherwise. The pure helpers still export; the server half throws
"shell-link needs Electron: this half only runs inside the desktop app".

The rules, each a mirror of the backend's with a comment saying so:
- `isLocalHost(hostname)` — a port of `backend/src/plugins/hosts.ts` (`localhost`/`*.localhost`, the trailing
  dot, 127/8, 0/8, 10/8, 172.16/12, 192.168/16, 169.254/16, 100.64/10, 198.18/15, 224/4, 240/4; `::1`, `::`,
  fc00::/7, fe80::/10, ff00::/8, and the IPv4-mapped and NAT64 forms judged as the v4 address they carry).
- `safeResponseHeaders(headers)` — the same allow-list as `backend/src/plugins/shell.ts` (content-type,
  content-length, etag, last-modified, link, retry-after, location, cache-control, date, `x-ratelimit-*`),
  lower-cased, values stripped of CR/LF and capped, `set-cookie`/`set-cookie2` never, at most 50. The input is
  duck-typed on `entries()`, so Electron's `Headers`, Node's, or a plain object all work.
  Values are capped at 8192, the backend's own MAX_HEADER_LENGTH: truncating harder than the backend
  would is a value the plugin never sees whole even though the backend would have passed it on.
- `requestHeadersFor(headers, { crossOrigin })` — drops the hop-by-hop set (`host`, `cookie`, `content-length`,
  `transfer-encoding`, `connection`, `upgrade`, `te`, `keep-alive`, `proxy-connection`) and *always* both
  credential headers: the shell attaches the site's login, and a plugin's own key must never ride along on a
  site fetch. Names must match the RFC token regex, values carrying CR/LF/NUL are dropped whole, values are
  capped at 4000, at most 40 headers. `crossOrigin` additionally sheds `origin` and `referer`, which name the
  origin the chain has just left.
- `nextHop(current, location, status, method)` → `{ url, method, dropBody }`. Resolves the Location against the
  current URL and refuses what the backend would refuse, naming the rule: a scheme that is not http(s), an
  https→http downgrade ('insecure redirect'), a local host ('local address'), an unreadable Location. 303 (and
  301/302 on POST) become a bodyless GET; a cross-origin 307/308 keeps the method and drops the body.

The server: `http.createServer` on 127.0.0.1 with a random port and a random 24-byte bearer token, compared
with `crypto.timingSafeEqual` on equal-length buffers. Every request must carry `authorization: Bearer <token>`
(401 otherwise), must be a POST (405), and its body is capped at 8 MB — a `/fetch` carries the plugin's own
request body inside a JSON envelope, and the backend's route accepts up to 1 MB of raw body, which escapes to
about six as `\uXXXX`. Over the cap the answer is `413 { error: 'request body is too large' }`, and the body is
**drained to its end first**: destroying the socket reaches the backend as a transport failure, and three of
those in a row make it deregister a healthy shell — and replying while the upload is still in flight is the same
failure wearing a different hat, since undici reports the early close as `fetch failed` and never reads the
reply. The drain gives up at 4× the cap, at which point the peer is not our backend having an off day. Routes:
- `POST /status {sites}` → `{ signedIn }` — a site is signed in when
  `session.fromPartition(partitionFor(site)).cookies.get({ url: site })` comes back non-empty.
- `POST /signin {site}` → opens the sign-in window.
- `POST /fetch {url, init}` → `{ status, headers, body, truncated }`. The first URL goes through the same
  http(s)/not-local gate; `init.method` must be in an allow-list and `init.body` must be text (and absent on a
  GET or HEAD). Up to 5 hops through `nextHop`, one 20 s timeout for the whole chain rather than one per hop,
  a 4 MB byte cap on the body (`content-length` is a claim, not a limit), and `content-length` in the reply set
  to what we actually hand over.

  **The fetch is `net.request`, not `session.fetch`.** Electron 33's `session.fetch` does not implement
  `redirect: 'manual'`: its net-fetch never registers a handler for the underlying request's `redirect` event,
  so the redirect is cancelled and the promise rejects with "Redirect was cancelled" on *any* 3xx. A hop loop
  built on it never runs a second hop — every redirecting site fetch just fails. `net.request({ session, url,
  method, redirect: 'manual', credentials: 'include' })` emits the event and lets us decide hop by hop.
  `followRedirect()` reuses the request it is called on, so it cannot change the method or drop the body: a hop
  that needs either (a 303, a POST through a 301/302, a cross-origin 307/308) aborts and issues a fresh request
  instead. Both kinds count against the same budget of 5.

  Two rules hold this together. Anything that runs as a listener (the header loop, the data handler, the
  delivery) settles the promise on failure rather than throwing, because a throw out of an emitter has nobody
  to catch it and leaves the promise pending for good. And every place that aborts a request whose answer is
  already known settles *first* and aborts second: `abort()` makes the net layer emit `'abort'`, whose handler
  rejects, and aborting first loses the answer to that rejection the moment the event arrives synchronously.
  A test fake that emits `'abort'` synchronously pins the ordering; the rest of the fakes emit it a tick later,
  the way Electron does.
Errors come back as `{ error }` with the message only, never a stack: this crosses a process boundary. The
*status* matters as much as the message — the backend counts 5xx towards deciding the shell has died, and three
of those deregister a healthy one — so everything the request itself could have caused (a bad url, a refused
hop, a timeout, a disallowed method, a non-text body, a site that would not answer) is a 400, and only an
unexpected internal fault is a 500. `close()` calls `closeAllConnections()` as well as `close()`: the backend holds its sockets open, and
`close()` alone would wait for them.

The three routes are injectable (`startShellLink({ log, impl })`) and so is the network layer (`fetchWithSite`
takes the request factory): only the defaults need Electron, so the server — token, method, routing, the 413,
the error shape — and the whole hop loop are testable in plain Node.

`close()` says goodbye before it shuts down: `POST /api/shell/goodbye { token }` on the backend (loopback only,
and the token must match the registered one) hands the link straight back, so a quitting shell does not leave
the backend offering sign-in and sending fetches to a dead port until three of them have failed. Best effort,
on a 1 s leash — we are quitting. `close()` also sets `hungUp` on the link, which `hello` checks before each
attempt: its retry loop can still be part-way through its ten seconds when the app quits, and the link it would
hand over is already closed.

Known limitation, recorded in the design doc too: a fetch made here cannot pin DNS, because Chromium's net
stack has no lookup hook. For a declared site the name-based `isLocalHost` check, re-applied on every
hop, is the only defence against DNS rebinding; the backend's plain fetches resolve once and dial what they
checked.

The sign-in window is `new BrowserWindow({ width: 1000, height: 760, title: 'Sign in — <host>',
webPreferences: { partition, contextIsolation: true, sandbox: true, nodeIntegration: false, devTools: false } })`
with the menu bar hidden and **no preload** — nothing of the app's runs in a page the user is typing a password
into. The site's session gets `setPermissionRequestHandler(cb => cb(false))` and `setPermissionCheckHandler(() =>
false)` first: a sign-in page has no business with the camera, the microphone, the user's location or
notifications. One window per partition, kept in a Map and cleared on `closed`, so a second Sign in click raises
the window that is already open rather than stacking another — pointing it back at the site first if the user
has wandered off somewhere else in the meantime.
`setWindowOpenHandler` denies the popup and loads the URL in the same window when it is http(s); `will-navigate`
blocks anything that is not http(s). Navigation is deliberately *not* pinned to the site's origin: an OAuth flow
hops to an identity provider and back, and a window that refuses the hop is a window nobody can sign in with.

after the last of them). Running every minute, it logs only when the answer *changes* — the first failure, and
the first success after one — rather than a line a minute forever. `stopBackend()` also closes the link.

```ts
// frontend/src/plugins/route.ts
import type { api as Api } from '../api';
import type { FeedMessage, PluginInfo, TokenInfo } from '../types';

export type Manifest = NonNullable<PluginInfo['manifest']>;

/** One field of a plugin's settings form, as the plugin declares it through `ot.settings.schema`. */
export interface SettingField {
  key: string;
  label: string;
  type: 'text' | 'number' | 'toggle' | 'secret';
  default?: unknown;
}

export interface PluginCall {
  method: string;
  args: unknown[];
}

export interface PluginContext {
  id: string;
  manifest: Manifest;
  api: typeof Api;
  actions: {
    openToken: (a: string) => void;
    jump: (msgId: string) => void;
    buy: (a: string) => void;
    research: (a: string) => void;
    copy: (t: string) => void;
    notify: (title: string, body: string) => void;
  };
  snapshot: { messages: () => FeedMessage[]; tokens: () => Record<string, TokenInfo> };
  ui: {
    setTitle: (t: string) => void;
    setSubtitle: (t: string) => void;
    badge: (n: number | null) => void;
    /** the settings form the plugin declared; the host keeps it, the backend never sees it */
    setSchema: (s: SettingField[]) => void;
  };
}

const need = (ctx: PluginContext, p: Manifest['permissions'][number]) => {
  if (!ctx.manifest.permissions.includes(p)) throw new Error(`this plugin did not ask for the ${p} permission`);
};

const str = (v: unknown, what: string, max = 4000): string => {
  if (typeof v !== 'string' || !v) throw new Error(`${what} must be a non-empty string`);
  return v.slice(0, max);
};

/** `str` where the empty string is a value of its own — clearing a title rather than an error. */
const strOrEmpty = (v: unknown, what: string, max: number): string => {
  if (typeof v !== 'string') throw new Error(`${what} must be a string`);
  return v.slice(0, max);
};

/** Look a plugin's key up as data: `constructor` and `__proto__` are misses, not inherited members. */
const own = <T>(obj: Record<string, T>, key: string): T | null => (Object.prototype.hasOwnProperty.call(obj, key) ? obj[key] : null);

const FIELD_TYPES = ['text', 'number', 'toggle', 'secret'];
const KEY_RE = /^[a-z0-9_-]{1,40}$/i;
/** keys that would reach through a plain settings object instead of sitting in it */
const RESERVED_KEYS = ['__proto__', 'constructor', 'prototype'];

/** A settings form is small, flat and declared in full: anything else is refused rather than half-shown. */
const settingsSchema = (v: unknown): SettingField[] => {
  if (!Array.isArray(v) || v.length > 20) throw new Error('the settings schema must be an array of at most 20 fields');
  const seen = new Set<string>();
  return v.map((raw) => {
    const f = (raw ?? {}) as Record<string, unknown>;
    if (typeof f.key !== 'string' || !KEY_RE.test(f.key)) throw new Error(`settings field key ${JSON.stringify(f.key)} must be 1-40 letters, digits, _ or -`);
    if (RESERVED_KEYS.includes(f.key.toLowerCase())) throw new Error(`settings field key ${f.key} is reserved`);
    if (seen.has(f.key)) throw new Error(`settings field key ${f.key} appears twice`);
    seen.add(f.key);
    if (typeof f.label !== 'string' || !f.label || f.label.length > 60) throw new Error(`settings field ${f.key} needs a label of at most 60 characters`);
    if (typeof f.type !== 'string' || !FIELD_TYPES.includes(f.type)) throw new Error(`settings field ${f.key} has an unknown type; use ${FIELD_TYPES.join(', ')}`);
    const field: SettingField = { key: f.key, label: f.label, type: f.type as SettingField['type'] };
    if ('default' in f) field.default = f.default;
    return field;
  });
};

/** Only what the proxy takes, and only as strings: a plugin cannot smuggle a Request, a stream or credentials through. */
const fetchInit = (v: unknown): { method?: string; headers?: Record<string, string>; body?: string } => {
  const o = (v ?? {}) as Record<string, unknown>;
  const init: { method?: string; headers?: Record<string, string>; body?: string } = {};
  if (typeof o.method === 'string') init.method = o.method.slice(0, 20);
  if (o.headers && typeof o.headers === 'object') {
    const headers: Record<string, string> = {};
    for (const [k, val] of Object.entries(o.headers as Record<string, unknown>)) if (typeof val === 'string') headers[k] = val;
    init.headers = headers;
  }
  if (typeof o.body === 'string') init.body = o.body;
  return init;
};

/** One bridge call from a plugin → one app-side effect. Pure apart from what it calls on `ctx`; permissions live here and nowhere else. */
export async function routeCall(ctx: PluginContext, call: PluginCall): Promise<unknown> {
  const [a0, a1] = call.args ?? [];
  switch (call.method) {
    // newest first, as the feed itself is ordered; messages the feed hides (blacklisted authors, bots the
    // bot policy refuses) are not the plugin's to read, so they never appear here
    case 'feed.messages': {
      const o = (a0 ?? {}) as { chat?: string; limit?: number };
      let list = ctx.snapshot.messages().filter((m) => !m.hidden);
      if (o.chat) list = list.filter((m) => m.chatName === o.chat || m.chatId === o.chat);
      return list.slice(0, Math.min(500, Math.max(1, Number(o.limit) || 100)));
    }
    case 'feed.tokens':
      return Object.values(ctx.snapshot.tokens()).slice(0, 2000);
    case 'feed.token':
      return own(ctx.snapshot.tokens(), str(a0, 'address', 120));
    case 'feed.post':
      need(ctx, 'feed:write');
      return ctx.api.pluginPost(ctx.id, a0);
    case 'feed.patch':
      need(ctx, 'feed:write');
      return ctx.api.pluginPatch(ctx.id, str(a0, 'id', 200), a1 ?? {});
    case 'fetch':
      return ctx.api.pluginFetch(ctx.id, str(a0, 'url', 2048), fetchInit(a1));
    case 'sites.status': {
      const site = str(a0, 'site', 200);
      const s = await ctx.api.pluginSites(ctx.id);
      return { signedIn: s.signedIn.includes(site), available: s.available };
    }
    case 'sites.signIn':
      return ctx.api.pluginSignIn(ctx.id, str(a0, 'site', 200));
    case 'storage.get':
      need(ctx, 'storage');
      return own(await ctx.api.pluginStorage(ctx.id), str(a0, 'key', 100));
    case 'storage.set':
      need(ctx, 'storage');
      return ctx.api.pluginStorageSet(ctx.id, str(a0, 'key', 100), a1);
    case 'storage.remove':
      need(ctx, 'storage');
      return ctx.api.pluginStorageSet(ctx.id, str(a0, 'key', 100), undefined);
    case 'settings.schema':
      ctx.ui.setSchema(settingsSchema(a0));
      return null;
```

```ts
// frontend/src/plugins/bridge.ts
/** The `ot` object as the plugin sees it. Every call is a postMessage round trip; events flow the other way. Append-only within api 1. */
export const BRIDGE_SRC = String.raw`
(() => {
  const pending = new Map();
  let seq = 0;
  const listeners = { message: new Set(), token: new Set() };
  const call = (method, ...args) => new Promise((resolve, reject) => {
    const id = ++seq;
    pending.set(id, { resolve, reject });
    parent.postMessage({ ot: 1, kind: 'call', id, method, args }, '*');
  });
  window.addEventListener('message', (e) => {
    if (e.source !== parent) return;
    const d = e.data;
    if (!d || d.ot !== 1) return;
    if (d.kind === 'result') {
      const p = pending.get(d.id);
      if (!p) return;
      pending.delete(d.id);
      d.error ? p.reject(new Error(d.error)) : p.resolve(d.value);
    } else if (d.kind === 'event') {
      const set = Object.prototype.hasOwnProperty.call(listeners, d.name) ? listeners[d.name] : null;
      for (const fn of set || []) { try { fn(d.data); } catch (err) { call('log', 'error', 'listener: ' + (err && err.message || err)).catch(() => {}); } }
    }
  });
  const on = (name) => (fn) => { listeners[name].add(fn); return () => listeners[name].delete(fn); };
  // { status, headers, body, truncated } from the proxy; truncated is true when the body hit the cap
  const withHelpers = (r) => ({ ...r, text: () => r.body, json: () => JSON.parse(r.body) });
  const F = Object.freeze;
  window.ot = F({
    api: 1,
    feed: F({
      onMessage: on('message'), onToken: on('token'),
      messages: (o) => call('feed.messages', o || {}), tokens: () => call('feed.tokens'), token: (a) => call('feed.token', a),
      post: (m) => call('feed.post', m), patch: (id, p) => call('feed.patch', id, p),
    }),
    fetch: (url, init) => call('fetch', url, init || {}).then(withHelpers),
    sites: F({ status: (s) => call('sites.status', s), signIn: (s) => call('sites.signIn', s) }),
    storage: F({ get: (k) => call('storage.get', k), set: (k, v) => call('storage.set', k, v), remove: (k) => call('storage.remove', k) }),
    settings: F({ schema: (s) => call('settings.schema', s), get: () => call('settings.get') }),
    ui: F({ root: document.body, setTitle: (t) => call('ui.setTitle', t), setSubtitle: (t) => call('ui.setSubtitle', t), badge: (n) => call('ui.badge', n) }),
    actions: F({
      openToken: (a) => call('actions.openToken', a), jump: (id) => call('actions.jump', id), buy: (a) => call('actions.buy', a),
      research: (a) => call('actions.research', a), copy: (t) => call('actions.copy', t), notify: (t, b) => call('actions.notify', t, b),
    }),
    log: (...a) => call('log', 'info', ...a),
  });
  window.addEventListener('error', (e) => call('log', 'error', String(e.message || e)).catch(() => {}));
  window.addEventListener('unhandledrejection', (e) => call('log', 'error', String(e.reason && e.reason.message || e.reason)).catch(() => {}));
})();
`;
```

Details the bridge has to get right: it is a `String.raw` template, so **no backtick and no `${`** may appear inside it (one backtick in a comment ends the literal); the event name off the wire is looked up with `hasOwnProperty`, so an `event` named `__proto__` finds nothing instead of `Object.prototype`; a `result` for an id the bridge never sent is dropped; anything without `ot === 1`, or from a sender that is not `parent`, is not ours; the leaf objects are frozen too, not just `ot` itself; and the three fire-and-forget `call('log', …)` sites end in `.catch(() => {})`, or the bridge's own error reporting becomes the next unhandled rejection.

`frontend/src/plugins/bridge.test.ts` evaluates `BRIDGE_SRC` in jsdom with `parent.postMessage` stubbed (jsdom's top window is its own `parent`, so a reply is dispatched with `source: window`) and covers exactly those: the api version and one call per method, `text()`/`json()` over `{ status, headers, body, truncated }`, an error result becoming a rejection, foreign, unknown-id and wrong-sender results being ignored, the frozen leaves, and `onMessage`/`onToken` delivering until their unsubscribe is called.

`frontend/src/plugins/surface.test.ts` keeps the two halves honest: it pulls every `call('x.y'` out of `BRIDGE_SRC` and every `case 'x.y':` out of `route.ts` (read with `fs` — note that jsdom's `URL` global is not node's, so resolve the path with `fileURLToPath` first) and asserts the two sets are equal, with an explicit `ROUTER_ONLY` list for any router case deliberately off the surface (empty today). It also asserts `BRIDGE_SRC` holds no backtick and no interpolation. A method added to one side and not the other is a plugin call that throws or a case nothing can reach; this test is what notices.

```js
  "api": 1,
  "sites": [],
  "permissions": ["feed:write", "storage"],

};

// A public endpoint with no login: GitHub's zen quotes are tiny and harmless.
const URL = 'https://api.github.com/zen';

export default function main(ot) {
  ot.ui.setTitle('Hello feed');
  const root = ot.ui.root;
  root.innerHTML = '<div style="padding:12px;color:var(--muted)">waiting for the first item…</div>';
  let timer;
  const tick = async () => {
    try {
      const r = await ot.fetch(URL, { headers: { accept: 'text/plain' } });
      const text = (await r.text()).trim();
      const last = await ot.storage.get('last');
      if (text && text !== last) {

        await ot.feed.post({ id: String(Date.now()), chat: 'Zen', author: 'github', text, link: 'https://api.github.com/zen' });
        root.innerHTML = `<div style="padding:12px">${text}</div>` + root.innerHTML;
      }
    } catch (e) {
      ot.log('error', e.message);
    }
  };
  tick();
  timer = setInterval(tick, 60_000);
  return () => clearInterval(timer);
```
